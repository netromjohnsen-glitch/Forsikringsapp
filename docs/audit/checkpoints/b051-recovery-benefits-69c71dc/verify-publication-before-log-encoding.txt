def check_relative_target(context, target):
    resolved = (context/target.split('#')[0]).resolve()
    resolved.relative_to(ROOT)  # Never accept a target outside the repo.
    assert resolved.exists(), (context, target)
    return resolved

for path in [*OUT.glob('*.md'), ROOT/inputs['path']]:
    assert b'\r' not in path.read_bytes()
    context = original_path.parent if path == archive else path.parent
    for target in re.findall(r'\]\(([^)]+)\)', path.read_text()):
        if not target.startswith(('http:', 'https:', '#')):
            check_relative_target(context, target)
for target in link_proof['targets']:
    resolved = check_relative_target(original_path.parent, target['relative_link'])
    assert str(resolved.relative_to(ROOT)) == target['resolved_repo_path']
    revision = target['revision']
    assert sha(resolved.read_bytes()) == revision['sha256']
    if revision['type'] == 'COMMITTED_BASELINE':
        assert revision['commit'] == BASE
        assert sha(subprocess.check_output(['git', 'show', BASE+':'+target['resolved_repo_path']], cwd=ROOT)) == revision['sha256']
    else:
        assert revision['type'] == 'IMMUTABLE_UNCOMMITTED_CANDIDATE_ARTIFACT' and revision['baseline'] == BASE
        stopped = dict((p, h) for h, p in (line.split('  ', 1) for line in (OUT/'STOP_SHA256SUMS').read_text().splitlines()))
        assert stopped[target['resolved_repo_path']] == revision['sha256']
# Real negative checks: the old wrong context and a nonexistent target must fail.
for context, target in [(archive.parent, link_proof['targets'][0]['relative_link']),
                        (original_path.parent, '__B051_MISSING_LINK_TARGET_NEGATIVE_CONTROL__.json')]:
    try:
        check_relative_target(context, target)
    except AssertionError:
        pass
    else:
        raise AssertionError(('Negative link control unexpectedly accepted', context, target))
staged = subprocess.check_output(['git', 'diff', '--cached', '--name-only', '-z'], cwd=ROOT).decode().split('\0')[:-1]
if staged:
    assert len(staged) == len(expected) and set(staged) == set(expected)

        assert subprocess.check_output(['git', 'show', ':'+path], cwd=ROOT) == (ROOT/path).read_bytes(), path

subprocess.run(['git', 'diff', '--cached', '--check'], cwd=ROOT, check=True)
print('Publication integrity PASS: exact one-byte delta; retained complete gates + fresh115/115;48 immutable prior receipts;51 unique signatures;B05125/10/2/2;budget23/12 scope3/6;strict staged-set and both diff checks.')
